SELECT *
FROM employees
WHERE salary >
(
    SELECT AVG(salary)
    FROM employees
);

SELECT *
FROM products
WHERE unit_price >
(
    SELECT MIN(price)
    FROM orders
);

SELECT *
FROM customers
WHERE customer_id IN
(
    SELECT customer_id
    FROM orders
);

SELECT
    customer_name,
    (
        SELECT COUNT(*)
        FROM orders o
        WHERE o.customer_id = c.customer_id
    ) AS order_count
FROM customers c;

SELECT *
FROM employees e1
WHERE salary =
(
    SELECT MAX(salary)
    FROM employees e2
    WHERE e1.department = e2.department
);

SELECT EXISTS
(
    SELECT *
    FROM products
    WHERE name = 'Laptop'
) AS product_exists;
