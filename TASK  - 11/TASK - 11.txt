SELECT *
FROM employees e1
WHERE salary >
(
    SELECT AVG(salary)
    FROM employees e2
    WHERE e1.department = e2.department
);

SELECT *
FROM customers
WHERE customer_id IN
(
    SELECT customer_id
    FROM orders
    GROUP BY customer_id
    HAVING COUNT(*) > 2
);

SELECT *
FROM products
WHERE unit_price =
(
    SELECT MAX(unit_price)
    FROM products
);

SELECT
    customer_id,
    (
        SELECT SUM(quantity * price)
        FROM orders o2
        WHERE o1.customer_id = o2.customer_id
    ) AS total_order_value
FROM orders o1
GROUP BY customer_id;

SELECT *
FROM employees e
WHERE salary >
(
    SELECT salary
    FROM employees m
    WHERE m.emp_id = e.manager_id
);

SELECT EXISTS
(
    SELECT *
    FROM orders
    WHERE product_name = 'Laptop'
) AS product_exists;
